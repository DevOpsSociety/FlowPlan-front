# WBS ↔ 간트차트 ↔ 칸반보드 통합 구현 계획

> **목표**: WBS 테이블, 간트차트, 칸반보드가 단일 데이터 소스를 공유하여 실시간으로 동기화되는 시스템 구축

---

## 📋 개요

### 현재 상황
- WBS 테이블: `HierarchicalWBSTask` 타입 사용, `useState`로 로컬 상태 관리
- 간트차트: 커스텀 구현, WBS와 데이터 분리
- 칸반보드: 미구현

### 해결 방안
- **Zustand** 전역 상태 관리로 단일 데이터 소스 구축
- **gantt-task-react** 라이브러리 도입
- **Adapter 패턴**으로 뷰별 데이터 변환
- 기존 `HierarchicalWBSTask` 타입 유지 (OpenAPI Codegen 대비)

#### 1.2 Zustand 스토어 생성

**파일**: `shared/stores/taskStore.ts`

```typescript
import { create } from 'zustand';
import { devtools, persist } from 'zustand/middleware';
import type { HierarchicalWBSTask } from '@/shared/lib/mockData';

interface TaskStore {
  // 상태
  projectTasks: Record<string, HierarchicalWBSTask[]>;

  // 조회
  getTasks: (projectId: string) => HierarchicalWBSTask[];
  setTasks: (projectId: string, tasks: HierarchicalWBSTask[]) => void;

  // CRUD
  updateTask: (projectId: string, taskId: string, updates: Partial<HierarchicalWBSTask>) => void;
  addTask: (projectId: string, task: HierarchicalWBSTask, parentId?: string) => void;
  deleteTask: (projectId: string, taskId: string) => void;

  // 헬퍼
  findTaskById: (projectId: string, taskId: string) => HierarchicalWBSTask | null;
}

// 재귀 헬퍼 함수들
function findTaskRecursively(
  tasks: HierarchicalWBSTask[],
  taskId: string
): HierarchicalWBSTask | null {
  for (const task of tasks) {
    if (task.id === taskId) return task;
    if (task.subTasks?.length) {
      const found = findTaskRecursively(task.subTasks, taskId);
      if (found) return found;
    }
  }
  return null;
}

function updateTaskRecursively(
  tasks: HierarchicalWBSTask[],
  taskId: string,
  updates: Partial<HierarchicalWBSTask>
): HierarchicalWBSTask[] {
  return tasks.map((task) => {
    if (task.id === taskId) {
      const updatedTask = { ...task, ...updates };

      // status 변경 시 progress 자동 업데이트
      if (updates.status) {
        if (updates.status === 'done') updatedTask.progress = 100;
        else if (updates.status === 'in-progress' && task.progress === 0) {
          updatedTask.progress = 10;
        }
        else if (updates.status === 'todo') updatedTask.progress = 0;
      }

      // duration 변경 시 endDate 자동 계산
      if (updates.duration && updates.duration !== task.duration) {
        const startDate = new Date(task.startDate);
        const newEndDate = new Date(startDate);
        newEndDate.setDate(startDate.getDate() + updates.duration - 1);
        updatedTask.endDate = newEndDate.toISOString().split('T')[0];
      }

      return updatedTask;
    }

    if (task.subTasks?.length) {
      return {
        ...task,
        subTasks: updateTaskRecursively(task.subTasks, taskId, updates),
      };
    }

    return task;
  });
}

function deleteTaskRecursively(
  tasks: HierarchicalWBSTask[],
  taskId: string
): HierarchicalWBSTask[] {
  return tasks.filter((task) => {
    if (task.id === taskId) return false;
    if (task.subTasks?.length) {
      task.subTasks = deleteTaskRecursively(task.subTasks, taskId);
    }
    return true;
  });
}

function addSubTaskRecursively(
  tasks: HierarchicalWBSTask[],
  parentId: string,
  newTask: HierarchicalWBSTask
): HierarchicalWBSTask[] {
  return tasks.map((task) => {
    if (task.id === parentId) {
      return {
        ...task,
        subTasks: [...(task.subTasks || []), newTask],
      };
    }
    if (task.subTasks?.length) {
      return {
        ...task,
        subTasks: addSubTaskRecursively(task.subTasks, parentId, newTask),
      };
    }
    return task;
  });
}

export const useTaskStore = create<TaskStore>()(
  devtools(
    persist(
      (set, get) => ({
        projectTasks: {},

        getTasks: (projectId) => get().projectTasks[projectId] || [],

        setTasks: (projectId, tasks) =>
          set((state) => ({
            projectTasks: { ...state.projectTasks, [projectId]: tasks },
          })),

        updateTask: (projectId, taskId, updates) =>
          set((state) => ({
            projectTasks: {
              ...state.projectTasks,
              [projectId]: updateTaskRecursively(
                state.projectTasks[projectId] || [],
                taskId,
                updates
              ),
            },
          })),

        addTask: (projectId, task, parentId) =>
          set((state) => {
            const currentTasks = state.projectTasks[projectId] || [];

            if (!parentId) {
              // 최상위 작업 추가
              return {
                projectTasks: {
                  ...state.projectTasks,
                  [projectId]: [...currentTasks, task],
                },
              };
            }

            // 하위 작업 추가
            return {
              projectTasks: {
                ...state.projectTasks,
                [projectId]: addSubTaskRecursively(currentTasks, parentId, task),
              },
            };
          }),

        deleteTask: (projectId, taskId) =>
          set((state) => ({
            projectTasks: {
              ...state.projectTasks,
              [projectId]: deleteTaskRecursively(
                state.projectTasks[projectId] || [],
                taskId
              ),
            },
          })),

        findTaskById: (projectId, taskId) =>
          findTaskRecursively(get().projectTasks[projectId] || [], taskId),
      }),
      { name: 'flowplan-tasks' }
    )
  )
);
```

#### 1.3 Adapter 함수 구현

**파일**: `shared/adapters/taskAdapters.ts`

```typescript
import type { Task as GanttTask } from 'gantt-task-react';
import type { HierarchicalWBSTask } from '@/shared/lib/mockData';

/**
 * HierarchicalWBSTask를 gantt-task-react Task로 변환 (평면화)
 * 계층 구조를 project 필드로 표현
 */
export function toGanttTasks(tasks: HierarchicalWBSTask[]): GanttTask[] {
  const result: GanttTask[] = [];

  function flatten(taskList: HierarchicalWBSTask[], parentId?: string) {
    taskList.forEach((task) => {
      result.push({
        id: task.id,
        name: task.name,
        start: new Date(task.startDate),
        end: new Date(task.endDate),
        progress: task.progress,
        type: task.subTasks?.length ? 'project' : 'task',
        dependencies: task.dependencies,
        project: parentId,
      });

      // 하위 작업 재귀 처리
      if (task.subTasks?.length) {
        flatten(task.subTasks, task.id);
      }
    });
  }

  flatten(tasks);
  return result;
}

/**
 * gantt-task-react Task를 HierarchicalWBSTask 업데이트로 변환
 */
export function fromGanttTask(ganttTask: GanttTask): Partial<HierarchicalWBSTask> {
  const duration = Math.ceil(
    (ganttTask.end.getTime() - ganttTask.start.getTime()) / (1000 * 60 * 60 * 24)
  ) + 1;

  return {
    startDate: ganttTask.start.toISOString().split('T')[0],
    endDate: ganttTask.end.toISOString().split('T')[0],
    progress: ganttTask.progress,
    duration,
  };
}

/**
 * HierarchicalWBSTask를 평면 리스트로 변환 (칸반보드용)
 */
export function flattenTasks(tasks: HierarchicalWBSTask[]): HierarchicalWBSTask[] {
  const result: HierarchicalWBSTask[] = [];

  function flatten(taskList: HierarchicalWBSTask[]) {
    taskList.forEach((task) => {
      result.push(task);
      if (task.subTasks?.length) {
        flatten(task.subTasks);
      }
    });
  }

  flatten(tasks);
  return result;
}
```

```typescript
            tasks={ganttTasks}
            viewMode={viewMode}
            onDateChange={handleTaskChange}
            onProgressChange={handleProgressChange}
            onClick={handleTaskClick}
            locale="ko"
            listCellWidth="250px"
            columnWidth={
              viewMode === ViewMode.Day ? 60
              : viewMode === ViewMode.Week ? 80
              : viewMode === ViewMode.Month ? 120

            }

  );
```

```typescript
'use client';

import { useTaskStore } from '@/shared/stores/taskStore';
import { flattenTasks } from '@/shared/adapters/taskAdapters';
import type { HierarchicalWBSTask } from '@/shared/lib/mockData';
import { Card, CardHeader, CardTitle, CardContent } from '@/shared/ui/card';
import { Badge } from '@/shared/ui/badge';
import { Avatar, AvatarFallback } from '@/shared/ui/avatar';

interface KanbanBoardProps {
  projectId: string;
  onTaskSelect?: (taskId: string) => void;
  selectedTaskId?: string | null;
}

const statusConfig = {
  todo: { label: '할 일', color: 'bg-gray-100' },
  'in-progress': { label: '진행 중', color: 'bg-blue-100' },
  done: { label: '완료', color: 'bg-green-100' },
  blocked: { label: '차단됨', color: 'bg-red-100' },
} as const;

export function KanbanBoard({ projectId, onTaskSelect, selectedTaskId }: KanbanBoardProps) {
  const tasks = useTaskStore((state) => state.getTasks(projectId));
  const updateTask = useTaskStore((state) => state.updateTask);

  // 계층 구조를 평면화
  const flatTasks = flattenTasks(tasks);

  // 상태별로 그룹화
  const tasksByStatus = {
    todo: flatTasks.filter((t) => t.status === 'todo'),
    'in-progress': flatTasks.filter((t) => t.status === 'in-progress'),
    done: flatTasks.filter((t) => t.status === 'done'),
    blocked: flatTasks.filter((t) => t.status === 'blocked'),
  };

  const handleDragStart = (e: React.DragEvent, taskId: string) => {
    e.dataTransfer.setData('taskId', taskId);
  };

  const handleDragOver = (e: React.DragEvent) => {
    e.preventDefault(); // 드롭 허용
  };

  const handleDrop = (e: React.DragEvent, newStatus: HierarchicalWBSTask['status']) => {
    e.preventDefault();
    const taskId = e.dataTransfer.getData('taskId');
    updateTask(projectId, taskId, { status: newStatus });
  };

  return (
    <div className="space-y-4">
      <h3 className="text-lg font-semibold">칸반보드</h3>

      <div className="grid grid-cols-4 gap-4">
        {Object.entries(tasksByStatus).map(([status, statusTasks]) => {
          const config = statusConfig[status as keyof typeof statusConfig];

          return (
            <div
              key={status}
              className="flex flex-col"
              onDragOver={handleDragOver}
              onDrop={(e) => handleDrop(e, status as HierarchicalWBSTask['status'])}
            >
              {/* 컬럼 헤더 */}
              <div className={`p-4 rounded-t-lg ${config.color}`}>
                <h4 className="font-semibold">
                  {config.label} ({statusTasks.length})
                </h4>
              </div>

              {/* 작업 카드들 */}
              <div className="flex-1 p-2 bg-muted/30 rounded-b-lg space-y-2 min-h-[400px]">
                {statusTasks.map((task) => (
                  <Card
                    key={task.id}
                    className={`cursor-pointer hover:shadow-md transition-shadow ${
                      selectedTaskId === task.id ? 'ring-2 ring-primary' : ''
                    }`}
                    draggable
                    onDragStart={(e) => handleDragStart(e, task.id)}
                    onClick={() => onTaskSelect?.(task.id)}
                  >
                    <CardContent className="p-4">
                      <div className="space-y-2">
                        {/* 작업명 */}
                        <h5 className="font-medium text-sm">{task.name}</h5>

                        {/* 진행률 */}
                        <div className="flex items-center space-x-2">
                          <div className="flex-1 bg-gray-200 rounded-full h-2">
                            <div
                              className="bg-primary h-2 rounded-full transition-all"
                              style={{ width: `${task.progress}%` }}
                            />
                          </div>
                          <span className="text-xs text-muted-foreground">
                            {task.progress}%
                          </span>
                        </div>

                        {/* 담당자 */}
                        <div className="flex items-center justify-between">
                          <div className="flex items-center space-x-2">
                            <Avatar className="h-6 w-6">
                              <AvatarFallback className="text-xs">
                                {task.assignee.substring(0, 2)}
                              </AvatarFallback>
                            </Avatar>
                            <span className="text-xs text-muted-foreground">
                              {task.assignee}
                            </span>
                          </div>

                          {/* 기한 */}
                          <span className="text-xs text-muted-foreground">
                            {task.endDate}
                          </span>
                        </div>
                      </div>
                    </CardContent>
                  </Card>
                ))}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

```typescript
  }, [projectId, tasks.length, setTasks]);

  return (

      {/* 메인 영역 */}
      <div className="flex-1 overflow-auto">
        <Tabs defaultValue="wbs" className="w-full p-6">
          <TabsList>
            <TabsTrigger value="wbs">WBS</TabsTrigger>
            <TabsTrigger value="gantt">간트차트</TabsTrigger>
            <TabsTrigger value="kanban">칸반보드</TabsTrigger>
          </TabsList>

          <TabsContent value="wbs" className="mt-6">
            <WBSTable
              projectId={projectId}
              onTaskSelect={setSelectedTaskId}
              selectedTaskId={selectedTaskId}
            />
          </TabsContent>

          <TabsContent value="gantt" className="mt-6">
            <GanttChartView
              projectId={projectId}
              onTaskSelect={setSelectedTaskId}
              selectedTaskId={selectedTaskId}
            />
          </TabsContent>

          <TabsContent value="kanban" className="mt-6">
            <KanbanBoard
              projectId={projectId}
              onTaskSelect={setSelectedTaskId}
              selectedTaskId={selectedTaskId}
            />
          </TabsContent>
        </Tabs>
      </div>

      {/* 사이드 패널 */}
      {selectedTaskId && (
        <div className="w-96 border-l">
          <TaskDetailPanel
            projectId={projectId}
            taskId={selectedTaskId}
            onClose={() => setSelectedTaskId(null)}
          />
        </div>
      )}
    </div>
  );
}
```

### Phase 3: 실시간 동기화 (선택)

#### 3.1 Spring WebSocket 통합 준비

**파일**: `shared/hooks/useRealtimeSync.ts` (신규)

```typescript
import { useEffect, useRef } from 'react';
import { Client } from '@stomp/stompjs';
import SockJS from 'sockjs-client';
import { useTaskStore } from '@/shared/stores/taskStore';
import type { HierarchicalWBSTask } from '@/shared/lib/mockData';

interface TaskUpdateMessage {
  type: 'TASK_UPDATED' | 'TASK_ADDED' | 'TASK_DELETED';
  projectId: string;
  taskId: string;
  updates?: Partial<HierarchicalWBSTask>;
  task?: HierarchicalWBSTask;
  userId: string;
}

export function useRealtimeSync(projectId: string, userId: string) {
  const clientRef = useRef<Client>();
  const updateTask = useTaskStore((state) => state.updateTask);
  const addTask = useTaskStore((state) => state.addTask);
  const deleteTask = useTaskStore((state) => state.deleteTask);

  useEffect(() => {
    const socket = new SockJS(`${process.env.NEXT_PUBLIC_API_URL}/ws`);
    const client = new Client({
      webSocketFactory: () => socket,

      onConnect: () => {
        console.log('WebSocket 연결됨');

        // 프로젝트별 토픽 구독
        client.subscribe(`/topic/project/${projectId}/tasks`, (message) => {
          const update: TaskUpdateMessage = JSON.parse(message.body);

          // 본인이 발생시킨 이벤트는 무시 (이미 로컬 업데이트됨)
          if (update.userId === userId) return;

          // 다른 사용자의 변경사항 반영
          switch (update.type) {
            case 'TASK_UPDATED':
              if (update.updates) {
                updateTask(projectId, update.taskId, update.updates);
              }
              break;

            case 'TASK_ADDED':
              if (update.task) {
                addTask(projectId, update.task);
              }
              break;

            case 'TASK_DELETED':
              deleteTask(projectId, update.taskId);
              break;
          }
        });
      },

      onDisconnect: () => {
        console.log('WebSocket 연결 해제됨');
      },

      onStompError: (frame) => {
        console.error('STOMP 에러:', frame);
      },
    });

    client.activate();
    clientRef.current = client;

    return () => {
      client.deactivate();
    };
  }, [projectId, userId, updateTask, addTask, deleteTask]);

  // 메시지 전송 함수
  const sendUpdate = (
    type: TaskUpdateMessage['type'],
    taskId: string,
    data: Partial<HierarchicalWBSTask> | HierarchicalWBSTask
  ) => {
    if (!clientRef.current?.connected) return;

    const message: TaskUpdateMessage = {
      type,
      projectId,
      taskId,
      userId,
      ...(type === 'TASK_UPDATED' ? { updates: data } : { task: data as HierarchicalWBSTask }),
    };

    clientRef.current.publish({
      destination: `/app/project/${projectId}/task/${type.toLowerCase()}`,
      body: JSON.stringify(message),
    });
  };

  return { sendUpdate };
}
```

**사용 예시**:
```typescript
// 프로젝트 페이지에서
const { sendUpdate } = useRealtimeSync(projectId, currentUser.id);

// Zustand 액션을 래핑
const updateTaskWithSync = (taskId: string, updates: Partial<HierarchicalWBSTask>) => {
  // 로컬 업데이트
  useTaskStore.getState().updateTask(projectId, taskId, updates);

  // 서버로 브로드캐스트
  sendUpdate('TASK_UPDATED', taskId, updates);
};
```
